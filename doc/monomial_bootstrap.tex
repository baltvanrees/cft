\section{Setup and Definitions}

Let $\mathbb{H}_- := \{w \in \mathbb{C} : \operatorname{Re}(w) < 0\}$ denote the open left half-plane and $\mathbb{H}_-^2 := \mathbb{H}_- \times \mathbb{H}_-$ the corresponding polydisk in exponential radial coordinates $(w_l, w_r)$, related to $(q_l, q_r) = (e^{w_l}, e^{w_r}) \in (\mathbb{D} \setminus \{0\})^2$ and the standard cross-ratio variables $(z, \bar{z})$ via
\begin{equation}
  z(w) := \frac{4 e^w}{(1 + e^w)^2}, \qquad
  \rho(z) := \frac{z}{\bigl(1 + \sqrt{1 - z}\bigr)^2}.
\end{equation}

\begin{definition}[Monomial Four-Point Function]
A \emph{monomial four-point datum} consists of a family of non-negative Borel measures $\{\mu_J\}_{J \in \mathbb{Z}}$ on $\mathbb{R}$ satisfying:
\begin{enumerate}
  \item \emph{Unitarity support}: $\mu_J(\{h \in \mathbb{R} : h < 0 \text{ or } h + J < 0\}) = 0$ for all $J \in \mathbb{Z}$;
  \item \emph{Bose symmetry}: $\mu_J = 0$ for all odd $J \in 2\mathbb{Z} + 1$;
  \item \emph{Absolute convergence}: for all $\sigma_l < 0$ and $\sigma_r < 0$, the real monomial integrals $e^{h \sigma_l + (h + J)\sigma_r}$ are $\mu_J$-integrable and
  \begin{equation}
    \mathcal{G}(\sigma_l, \sigma_r) := \sum_{J \in \mathbb{Z}} \int_{\mathbb{R}} e^{h \sigma_l + (h + J)\sigma_r} \, d\mu_J(h) < \infty.
  \end{equation}
\end{enumerate}
For $(w_l, w_r) \in \mathbb{H}_-^2$ and $w \in \mathbb{H}_-$, the holomorphic four-point function $\mathcal{G} : \mathbb{H}_-^2 \to \mathbb{C}$ and the spin-$J$ Laplace transforms $L_J : \mathbb{H}_- \to \mathbb{C}$ are defined by
\begin{equation}
  \mathcal{G}(w_l, w_r) := \sum_{J \in \mathbb{Z}} e^{J w_r} L_J(w_l + w_r), \qquad
  L_J(w) := \int_{\mathbb{R}} e^{h w} \, d\mu_J(h).
\end{equation}
\end{definition}

On the $t$-channel strip $\mathcal{S}_\pi := \{(w_l, w_r) \in \mathbb{H}_-^2 : \operatorname{Im}(w_l), \operatorname{Im}(w_r) \in (-\pi, \pi)\}$, the crossing transformation maps $w \in \mathbb{H}_-$ with $\operatorname{Im}(w) \in (-\pi, \pi)$ to $w' := \log \rho(1 - z(w)) \in \mathbb{H}_-$. For external scalar scaling dimension $\Delta_\phi > 0$, define the single-variable crossing factor and the $t$-channel identity contribution by
\begin{equation}
  \mathcal{C}_{\Delta_\phi}(w) := 4^{\Delta_\phi} e^{\Delta_\phi w} (1 - e^w)^{-2\Delta_\phi}, \qquad
  \mathcal{G}_{\mathrm{id}}(w_l, w_r) := \mathcal{C}_{\Delta_\phi}(w_l) \, \mathcal{C}_{\Delta_\phi}(w_r).
\end{equation}

\begin{definition}[Crossing Symmetry and Twist Gap]
A monomial four-point datum $\{\mu_J\}_{J \in \mathbb{Z}}$ with external dimension $\Delta_\phi > 0$ is \emph{crossing-symmetric} on $\mathcal{S}_\pi$ if
\begin{equation}
  \label{eq:crossing-sym}
  \mathcal{G}(w_l, w_r) = \mathcal{C}_{\Delta_\phi}(w_l) \, \mathcal{C}_{\Delta_\phi}(w_r) \, \mathcal{G}(w_l', w_r') \qquad \forall (w_l, w_r) \in \mathcal{S}_\pi.
\end{equation}
It has a \emph{twist gap} $h_{\mathrm{gap}} > 0$ above the vacuum ($h = J = 0$) if $\{\mu_J\}_{J \in \mathbb{Z}}$ decomposes as $\mu_0 = \delta_0 + \tilde{\mu}_0$ and $\mu_J = \tilde{\mu}_J$ ($J \ne 0$) for a monomial four-point datum $\{\tilde{\mu}_J\}_{J \in \mathbb{Z}}$ satisfying $\tilde{\mu}_J(\{h \in \mathbb{R} : h < h_{\mathrm{gap}} \text{ or } h + J < h_{\mathrm{gap}}\}) = 0$ for all $J \in \mathbb{Z}$.
\end{definition}

\begin{theorem}[Monomial Decomposition and Crossing Symmetry of Conformal Four-Point Functions]
\label{thm:monomial-decomp}
Every reflection-positive, Euclidean crossing-symmetric conformal four-point correlation function of identical scalars of dimension $\Delta_\phi > 0$ in $d \ge 2$ dimensions ($\nu = (d-2)/2 \ge 0$) canonically defines a monomial four-point datum $\{\mu_J\}_{J \in \mathbb{Z}}$---obtained by pushing forward the conformal-block OPE measure $\mu(\Delta, l)$ weighted by $A_{n, j}(\Delta, l) \, c_{j, a}^{(\nu)}$ along $h(\Delta, l) = \frac{\Delta + n - j}{2} + a$ and summing over all $(n, j, a)$ with $0 \le a \le j$ and $j - 2a = J$---whose evaluation $\mathcal{G}(w_l, w_r)$ coincides with the analytically continued $s$-channel four-point function on $\mathbb{H}_-^2$ and satisfies the two-variable crossing symmetry relation~\eqref{eq:crossing-sym} on $\mathcal{S}_\pi$.
\end{theorem}

\section{Tauberian Analysis on the Integer Lattice}

\begin{proposition}[Fej\'er--Hermite Lattice Measure Estimate]
\label{prop:hermite-measure}
For every $N \in \mathbb{N}$, define
\begin{equation}
  C_N := (2N + 1)\Bigl(1 + \frac{1}{\pi}\Bigr) + \frac{1}{4}\Bigl(1 + (2N + 1)\Bigl(\frac{2\pi^2}{3} + \frac{4\pi}{3}\Bigr)\Bigr).
\end{equation}
Let $\mu, \nu$ be finite positive Borel measures on $\mathbb{R}$ with $\nu(\mathbb{R} \setminus \mathbb{Z}) = 0$, and suppose that for some $\varepsilon \ge 0$,
\begin{equation}
  \left|\int_{\mathbb{R}} e^{i t u} \, d(\mu - \nu)(u)\right| \le \varepsilon \qquad \forall t \in [0, 2\pi].
\end{equation}
Then for every $\psi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{u \in \mathbb{R}} \max\{|\psi(u)|, |\psi'(u)|, |\psi''(u)|\} \le M_\psi$,
\begin{equation}
  \left|\int_{\mathbb{R}} \psi(u) \, d\mu(u) - \int_{\mathbb{R}} \psi(u) \, d\nu(u)\right| \le C_N \, \varepsilon \, M_\psi.
\end{equation}
\end{proposition}

\begin{proposition}[Nevanlinna's Two-Constants Theorem on a Vertical Strip]
\label{prop:two-constants}
Let $\sigma_- < \sigma_+$ and define the strip harmonic measure of the real segment $[\sigma_-, \sigma_+]$ by $\omega(\sigma, 0) := 1$ and
\begin{equation}
  \omega(\sigma, t) := \frac{2}{\pi} \arctan\!\left(\frac{\sin\bigl(\pi(\sigma - \sigma_-)/(\sigma_+ - \sigma_-)\bigr)}{\sinh\bigl(\pi |t|/(\sigma_+ - \sigma_-)\bigr)}\right) \qquad (t \ne 0).
\end{equation}
If $E : \mathbb{C} \to \mathbb{C}$ is holomorphic on $\{s \in \mathbb{C} : \operatorname{Re}(s) \in (\sigma_-, \sigma_+)\}$, continuous and bounded by $M$ on $\{s \in \mathbb{C} : \operatorname{Re}(s) \in [\sigma_-, \sigma_+]\}$, and satisfies $|E(\sigma)| \le \delta$ for all $\sigma \in [\sigma_-, \sigma_+]$ with $0 < \delta \le M$, then
\begin{equation}
  \bigl|E(\sigma + i t)\bigr| \le M^{1 - \omega(\sigma, t)} \, \delta^{\omega(\sigma, t)} \qquad \forall \sigma \in [\sigma_-, \sigma_+], \; \forall t \in \mathbb{R}.
\end{equation}
In particular, for $A > 1$, $T > 0$, $[\sigma_-, \sigma_+] = [1 - 2A, -1]$, and $\vartheta(A, T) := \frac{2}{\pi}\arctan\bigl(1/\sinh(\frac{\pi T}{2(A-1)})\bigr) \in (0, 1)$, we have $\lim_{A \to \infty} \vartheta(A, T) = 1$ and $|E(-A + i t)| \le M^{1 - \vartheta(A, T)} \delta^{\vartheta(A, T)}$ for all $t \in [-T, T]$.
\end{proposition}

\begin{theorem}[Quantitative Laplace-to-$C_c^2$ Tauberian Theorem on $\mathbb{Z}$]
\label{thm:laplace-tauberian}
Let $A > 1$, $0 < \delta \le M$, and $\vartheta_A := \vartheta(A, 2\pi) = \frac{2}{\pi}\arctan\bigl(1/\sinh(\frac{\pi^2}{A-1})\bigr)$. Let $\mu, \nu$ be positive Borel measures on $\mathbb{R}$ with $\nu(\mathbb{R} \setminus \mathbb{Z}) = 0$ such that $e^{(1 - 2A)u}$ and $e^{-u}$ are integrable with respect to both $\mu$ and $\nu$, and for all $\sigma \in [1 - 2A, -1]$,
\begin{equation}
  \int_{\mathbb{R}} e^{\sigma u} \, d\mu(u) + \int_{\mathbb{R}} e^{\sigma u} \, d\nu(u) \le M, \qquad
  \left|\int_{\mathbb{R}} e^{\sigma u} \, d\mu(u) - \int_{\mathbb{R}} e^{\sigma u} \, d\nu(u)\right| \le \delta.
\end{equation}
Then for every $N \in \mathbb{N}$ and every $\varphi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{u \in \mathbb{R}} \max\{|\varphi(u)|, |\varphi'(u)|, |\varphi''(u)|\} \le M_\varphi$,
\begin{equation}
  \left|\int_{\mathbb{R}} \varphi(u) \, d\mu(u) - \int_{\mathbb{R}} \varphi(u) \, d\nu(u)\right|
  \le C_N (A + 1)^2 e^{A N} M^{1 - \vartheta_A} \delta^{\vartheta_A} M_\varphi.
\end{equation}
Consequently, for any $0 < \gamma < \alpha$, choosing $A > 1$ with $\alpha \vartheta_A \ge \gamma$ upgrades any power-law Laplace error bound $\delta = C_0 J^{-\alpha}$ ($J \ge 1$) to a power-law $C_c^2$ pairing bound $C J^{-\gamma} M_\varphi$.
\end{theorem}

\section{Uniform Real-Axis Laplace Bounds and Quantitative Lightcone Bootstrap}

\begin{proposition}[$t$-Channel Twist-Gap Remainder Bound and Folded Spin Projection]
\label{prop:gap-and-proj}
Let $\{\mu_J\}_{J \in \mathbb{Z}}$ be a crossing-symmetric monomial four-point datum with external dimension $\Delta_\phi > 0$ and twist gap $h_{\mathrm{gap}} > 0$.
\begin{enumerate}
  \item For any $a, b < 0$, set $\sigma_{l,0} := \log(1 - \frac{\sqrt{2}}{4} e^{a}) < 0$ and $\sigma_{r,0} := \log(1 - \frac{\sqrt{2}}{4} e^{b}) < 0$. Then for all $(w_l, w_r) \in \mathbb{H}_-^2$ satisfying $\operatorname{Im}(w_l), \operatorname{Im}(w_r) \in [-\pi/2, \pi/2]$ and $\operatorname{Re}(w_l) \le a$, $\operatorname{Re}(w_r) \le b$,
  \begin{equation}
    \bigl|\mathcal{G}(w_l, w_r) - \mathcal{G}_{\mathrm{id}}(w_l, w_r)\bigr|
    \le 16^{\Delta_\phi} (1 - e^a)^{-2\Delta_\phi} e^{-h_{\mathrm{gap}} \sigma_{r,0}} \bigl(\mathcal{G}(\sigma_{l,0}, \sigma_{r,0}) - 1\bigr) \, |1 - e^{w_r}|^{2(h_{\mathrm{gap}} - \Delta_\phi)}.
  \end{equation}
  \item For every even spin $J = 2k \in 2\mathbb{Z}$ and every $(w_l, w_r) \in \mathbb{H}_-^2$,
  \begin{equation}
    L_{2k}(w_l + w_r) = \int_{-\pi/2}^{\pi/2} \frac{d\theta}{\pi} \, e^{-2k (w_r + i\theta)} \, \mathcal{G}(w_l - i\theta, w_r + i\theta).
  \end{equation}
\end{enumerate}
\end{proposition}

For $\Delta_\phi > 0$, even spin $J = 2k$ ($k \in \mathbb{N}$), $\sigma < 0$, and $\varphi : \mathbb{R} \to \mathbb{R}$, define the exact finite-spin $t$-channel identity Laplace transform $L_{\mathrm{id}, 2k}(\Delta_\phi; \sigma)$ and monomial pairing $\mathcal{P}_{\mathrm{id}, 2k}(\Delta_\phi; \varphi)$, as well as their universal $k \to \infty$ limits $L_{\mathrm{univ}}(\Delta_\phi; \sigma)$ and $\mathcal{P}_{\mathrm{univ}}(\Delta_\phi; \varphi)$, by
\begin{align}
  L_{\mathrm{id}, 2k}(\Delta_\phi; \sigma) &:= 2 \cdot 4^{2\Delta_\phi} e^{\Delta_\phi \sigma} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \frac{(2\Delta_\phi)_{n + 2k}}{(n + 2k)!} \, e^{n \sigma}, &
  L_{\mathrm{univ}}(\Delta_\phi; \sigma) &:= \frac{2 \cdot 4^{2\Delta_\phi}}{\Gamma(2\Delta_\phi)} \, e^{\Delta_\phi \sigma} (1 - e^\sigma)^{-2\Delta_\phi}, \\
  \mathcal{P}_{\mathrm{id}, 2k}(\Delta_\phi; \varphi) &:= 2 \cdot 4^{2\Delta_\phi} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \frac{(2\Delta_\phi)_{n + 2k}}{(n + 2k)!} \, \varphi(\Delta_\phi + n), &
  \mathcal{P}_{\mathrm{univ}}(\Delta_\phi; \varphi) &:= \frac{2 \cdot 4^{2\Delta_\phi}}{\Gamma(2\Delta_\phi)} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \, \varphi(\Delta_\phi + n).
\end{align}

\begin{proposition}[Uniform Real-Axis Laplace Bounds on Compact Negative Intervals]
\label{prop:uniform-laplace}
Let $\{\mu_J\}_{J \in \mathbb{Z}}$ be a crossing-symmetric monomial four-point datum with external dimension $\Delta_\phi > 1/2$ and twist gap $h_{\mathrm{gap}} > 0$ such that $2(h_{\mathrm{gap}} - \Delta_\phi) \ne -1$, and let $[\sigma_-, \sigma_+] \subset (-\infty, 0)$ be a compact interval.
\begin{enumerate}
  \item \emph{(Comparison with the exact finite-spin $t$-channel identity)} Set $\alpha_{\mathrm{id}} := \min(2 h_{\mathrm{gap}},\, 2\Delta_\phi - 1) > 0$. There exist constants $M_{\mathrm{bd}}, C_{\mathrm{bd}} > 0$ such that for all integers $k \ge 1$ and all $\sigma \in [\sigma_-, \sigma_+]$,
  \begin{align}
    (2k)^{1 - 2\Delta_\phi} L_{2k}(\sigma) + (2k)^{1 - 2\Delta_\phi} L_{\mathrm{id}, 2k}(\Delta_\phi; \sigma) &\le M_{\mathrm{bd}}, \\
    \Bigl|(2k)^{1 - 2\Delta_\phi} L_{2k}(\sigma) - (2k)^{1 - 2\Delta_\phi} L_{\mathrm{id}, 2k}(\Delta_\phi; \sigma)\Bigr| &\le C_{\mathrm{bd}} \, (2k)^{-\alpha_{\mathrm{id}}}.
  \end{align}
  \item \emph{(Comparison with the asymptotic universal limit)} Set $\alpha_{\mathrm{univ}} := \min(2 h_{\mathrm{gap}},\, 1,\, 2\Delta_\phi - 1) > 0$. There exist constants $M_{\mathrm{bd}}', C_{\mathrm{bd}}' > 0$ such that for all integers $k \ge 1$ and all $\sigma \in [\sigma_-, \sigma_+]$,
  \begin{align}
    (2k)^{1 - 2\Delta_\phi} L_{2k}(\sigma) + L_{\mathrm{univ}}(\Delta_\phi; \sigma) &\le M_{\mathrm{bd}}', \\
    \Bigl|(2k)^{1 - 2\Delta_\phi} L_{2k}(\sigma) - L_{\mathrm{univ}}(\Delta_\phi; \sigma)\Bigr| &\le C_{\mathrm{bd}}' \, (2k)^{-\alpha_{\mathrm{univ}}}.
  \end{align}
\end{enumerate}
\end{proposition}

\begin{theorem}[Quantitative Monomial Lightcone Bootstrap]
\label{thm:quantitative-large-spin}
Let $\{\mu_J\}_{J \in \mathbb{Z}}$ be the monomial four-point datum of a reflection-positive conformal four-point function in $d \ge 2$ dimensions with external dimension $\Delta_\phi > 1/2$ and twist gap $h_{\mathrm{gap}} > 0$, and let $N \in \mathbb{N}$.
\begin{enumerate}
  \item \emph{(Against the exact finite-spin $t$-channel identity)} For every target exponent $0 < \gamma < \min(2 h_{\mathrm{gap}},\, 2\Delta_\phi - 1)$, there exists a constant $C > 0$ such that for all integers $k \ge 1$ and all $\varphi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{h \in \mathbb{R}} \max\{|\varphi(h)|, |\varphi'(h)|, |\varphi''(h)|\} \le M_\varphi$,
  \begin{equation}
    \left|(2k)^{1 - 2\Delta_\phi} \int_{\mathbb{R}} \varphi(h) \, d\mu_{2k}(h)
      - (2k)^{1 - 2\Delta_\phi} \, \mathcal{P}_{\mathrm{id}, 2k}(\Delta_\phi; \varphi)\right|
    \le C \, (2k)^{-\gamma} \, M_\varphi.
  \end{equation}
  \item \emph{(Against the asymptotic universal double-twist limit)} For every target exponent $0 < \gamma < \min(2 h_{\mathrm{gap}},\, 1,\, 2\Delta_\phi - 1)$, there exists a constant $C' > 0$ such that for all integers $k \ge 1$ and all $\varphi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{h \in \mathbb{R}} \max\{|\varphi(h)|, |\varphi'(h)|, |\varphi''(h)|\} \le M_\varphi$,
  \begin{equation}
    \left|(2k)^{1 - 2\Delta_\phi} \int_{\mathbb{R}} \varphi(h) \, d\mu_{2k}(h)
      - \mathcal{P}_{\mathrm{univ}}(\Delta_\phi; \varphi)\right|
    \le C' \, (2k)^{-\gamma} \, M_\varphi.
  \end{equation}
\end{enumerate}
\end{theorem}
