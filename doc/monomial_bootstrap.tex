This note summarizes the main physically relevant lightcone and high-energy Tauberian theorems for conformal four-point functions formalized in Lean~4 in this repository, together with all axiomatic and structural assumptions on which they rest.

\section{Base Euclidean Axioms and Polydisk Continuation}

We consider the four-point correlation function of identical real scalar primary operators $\phi$ of scaling dimension $\Delta_\phi > 0$ in spacetime dimension $d = 2\nu + 2 \ge 2$ ($\nu = \frac{d-2}{2} \ge 0$). In radial coordinates $(\rho, \bar{\rho}) \in (\mathbb{D} \setminus \{0\})^2$, we pass to the universal cover by writing $(\rho_l, \rho_r) = (e^{w_l}, e^{w_r})$ with $(w_l, w_r) \in \mathbb{H}_-^2$, where $\mathbb{H}_- := \{w \in \mathbb{C} : \operatorname{Re} w < 0\}$. The cross-ratio map $z : \mathbb{H}_- \to \mathbb{C}$ and its inverse radial map on the twice-cut plane $\mathcal{C} := \mathbb{C} \setminus \bigl((-\infty, 0] \cup [1, \infty)\bigr)$ are
\begin{equation}
  z(w) := \frac{4 e^w}{(1 + e^w)^2}, \qquad
  \rho(z) := \frac{z}{\bigl(1 + \sqrt{1 - z}\bigr)^2}.
\end{equation}
On the first-sheet strip $\mathcal{S}_\pi := \{(w_l, w_r) \in \mathbb{H}_-^2 : \operatorname{Im} w_l, \operatorname{Im} w_r \in (-\pi, \pi)\}$, the $t$-channel crossing map $z \mapsto 1 - z$ lifts to $w \mapsto w' := \log \rho(1 - z(w)) \in \mathbb{H}_-$, with crossing factor $\mathcal{C}_{\Delta_\phi}(w) := 4^{\Delta_\phi} e^{\Delta_\phi w} (1 - e^w)^{-2\Delta_\phi}$.

\begin{assumption}[Base Euclidean OPE and Crossing Axioms]
\label{assump:base}
A reflection-positive, crossing-symmetric four-point function with twist gap $h_{\mathrm{gap}} > 0$ consists of:
\begin{enumerate}
  \item \emph{Positive unitary OPE measure}: a non-negative Borel measure $d\mu_{\mathrm{OPE}}(\Delta, \ell)$ on $\mathbb{R} \times \mathbb{N}$ supported on unitary twists $\tau = \Delta - \ell \ge 0$ and even spins $\ell \in 2\mathbb{N}$, decomposing into a unit vacuum atom at $(\Delta, \ell) = (0, 0)$ plus a non-identity measure supported on half-twist $h := \frac{\Delta - \ell}{2} \ge h_{\mathrm{gap}} > 0$;
  \item \emph{Euclidean conformal blocks}: for each $(\Delta, \ell)$, the canonical Euclidean conformal block $G_{\Delta, \ell}(r e^{i\theta}) = \sum_{n=0}^\infty r^{\Delta + n} \sum_{j} A_{n, j}(\Delta, \ell) \, C_j^{(\nu)}(\cos\theta)$ (\texttt{conformalBlock} in \texttt{Blocks.lean}) has measurable non-negative descendant coefficients $A_{n, j}(\Delta, \ell) \ge 0$ (normalized by $A_{0, \ell} = 1$, vanishing for odd $n$, and supported on $j \in \{\ell + n, \ell + n - 2, \dots, \max(\ell - n, (\ell + n) \bmod 2)\}$) and is radially summable on $r \in (0, 1)$;\footnote{The existence (\texttt{exists\_conformalBlock} in \texttt{Blocks.lean}) and measurability (\texttt{measurable\_conformalBlock\_A} in \texttt{Corr.lean}) of $G_{\Delta, \ell}$ are stated as \texttt{sorry}'d theorems rather than carried around inside \texttt{FourPointData}, so that \texttt{FourPointData} records only the physical OPE measure $d\mu_{\mathrm{OPE}}$, its unitarity and even-spin support, Euclidean OPE convergence, and Euclidean crossing symmetry.}
  \item \emph{Euclidean OPE convergence and crossing symmetry}: the OPE integral $\mathcal{G}_{\mathrm{eucl}}(\rho) := \int G_{\Delta, \ell}(\rho) \, d\mu_{\mathrm{OPE}}(\Delta, \ell)$ converges absolutely on $0 < |\rho| < 1$ and satisfies $|z|^{-2\Delta_\phi} \mathcal{G}_{\mathrm{eucl}}(\rho(z)) = |1 - z|^{-2\Delta_\phi} \mathcal{G}_{\mathrm{eucl}}(\rho(1 - z))$ for all $z \in \mathcal{C}$.
\end{enumerate}
\end{assumption}

Under Assumption~\ref{assump:base} alone (formalized with zero \texttt{sorry}'s in \texttt{Corr\_pdisk.lean}), expanding $C_j^{(\nu)}(\cos\theta)$ into $(\rho_l, \rho_r) = (e^{w_l}, e^{w_r})$ monomials with non-negative coefficients $c_{j, a}^{(\nu)} = \frac{(\nu)_a (\nu)_{j - a}}{a! (j - a)!} \ge 0$ proves that $\mathcal{G}_{\mathrm{eucl}}$ continues to a jointly holomorphic function $\mathcal{G} : \mathbb{H}_-^2 \to \mathbb{C}$ satisfying the two-variable crossing relation on $\mathcal{S}_\pi$:
\begin{equation}
  \label{eq:two-var-crossing}
  \mathcal{G}(w_l, w_r) = \mathcal{C}_{\Delta_\phi}(w_l) \, \mathcal{C}_{\Delta_\phi}(w_r) \, \mathcal{G}(w_l', w_r') \qquad \forall (w_l, w_r) \in \mathcal{S}_\pi.
\end{equation}

\section{Quantitative Monomial Lightcone Bootstrap (\texorpdfstring{$\ell \to \infty$}{l -> infty})}

Expanding each conformal block $G_{\Delta, \ell}(w_l, w_r)$ into powers $e^{h w_l + (h + J) w_r}$ pushes $d\mu_{\mathrm{OPE}}(\Delta, \ell)$ forward to a family of non-negative Borel measures $\{\mu_J\}_{J \in \mathbb{Z}}$ on $[0, \infty)$ supported on even spins $J = 2k$ (\texttt{MonomialDecomp.lean}, zero \texttt{sorry}'s), so that
\begin{equation}
  \mathcal{G}(w_l, w_r) = \sum_{k=0}^\infty e^{2k w_r} \int_{[0, \infty)} e^{h(w_l + w_r)} \, d\mu_{2k}(h).
\end{equation}

\begin{theorem}[Quantitative Monomial Lightcone Bootstrap --- \texttt{QuantitativeLargeSpin.lean}]
\label{thm:monomial-bootstrap}
Suppose Assumption~\ref{assump:base} holds with $\nu \ge 0$, external dimension $\Delta_\phi > 1/2$, and twist gap $h_{\mathrm{gap}} > 0$. Fix $N \in \mathbb{N}$ and any target exponent $0 < \gamma < \min\bigl(2 h_{\mathrm{gap}},\, 2\Delta_\phi - 1\bigr)$. Then there exists $C_N > 0$ such that for all integers $k \ge 1$ and all $\varphi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{h \in \mathbb{R}} \max\{|\varphi(h)|, |\varphi'(h)|, |\varphi''(h)|\} \le M_\varphi$,
\begin{equation}
  \label{eq:monomial-bootstrap}
  \left|(2k)^{1 - 2\Delta_\phi} \int_{\mathbb{R}} \varphi(h) \, d\mu_{2k}(h)
    - (2k)^{1 - 2\Delta_\phi} \, 2 \cdot 4^{2\Delta_\phi} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \frac{(2\Delta_\phi)_{n + 2k}}{(n + 2k)!} \, \varphi(\Delta_\phi + n)\right|
  \le C_N \, (2k)^{-\gamma} \, M_\varphi.
\end{equation}
\end{theorem}

\paragraph{Proof sketch and large-$k$ expansion of the universal part.}
By Bose symmetry ($\mu_J = 0$ for odd $J$), the real-axis Laplace transform $L_{2k}(\sigma) := \int e^{h\sigma} d\mu_{2k}(h)$ is recovered for $\sigma < 0$ by a folded Fourier integral over $\theta \in [-\pi/2, \pi/2]$ along $w_l = \sigma - \sigma_r - i\theta$, $w_r = \sigma_r + i\theta$, which stays strictly inside $\mathcal{S}_\pi$. Setting $\sigma_r = -1/(2k)$ and inserting~\eqref{eq:two-var-crossing} shows that on any compact interval $[\sigma_-, \sigma_+] \subset (-\infty, 0)$, the exact spin-$2k$ Laplace transform of the $t$-channel identity dominates $L_{2k}(\sigma)$ up to the purely non-universal error $O((2k)^{-\alpha})$ with $\alpha = \min(2 h_{\mathrm{gap}},\, 2\Delta_\phi - 1)$ (from the $t$-channel non-identity twist gap and the outer contour $[-\pi, -\pi/2] \cup [\pi/2, \pi]$). Applying Nevanlinna's Two-Constants Theorem on a wide vertical strip $\operatorname{Re} w \in [1 - 2A, -1]$ propagates this real-axis Laplace agreement to complex frequencies $\operatorname{Im} w \in [-2\pi, 2\pi]$ with harmonic-measure exponent $\vartheta_A = \frac{2}{\pi}\arctan(1/\sinh(\frac{\pi^2}{A-1})) \to 1$ as $A \to \infty$, and Fej\'er--Hermite interpolation on the lattice $\Delta_\phi + \mathbb{Z}$ upgrades bounded-frequency Fourier agreement to $C_c^2$ test functions. Theorem~\ref{thm:monomial-bootstrap} cleanly separates the non-universal $O((2k)^{-\gamma})$ dynamical corrections from the universal $t$-channel identity sum: expanding the universal finite-spin factor $(2k)^{1 - 2\Delta_\phi}\frac{(2\Delta_\phi)_{n + 2k}}{(n + 2k)!} = \frac{1}{\Gamma(2\Delta_\phi)} + O(k^{-1})$ to leading order at large $k$ (\texttt{exists\_const\_abs\_identityMonomialPairing\_sub\_universal\_le}) simplifies the subtracted sum in~\eqref{eq:monomial-bootstrap} to $\frac{2 \cdot 4^{2\Delta_\phi}}{\Gamma(2\Delta_\phi)} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \, \varphi(\Delta_\phi + n)$ plus a kinematic $O(k^{-1})$ truncation error, yielding exponent $0 < \gamma < \min(2 h_{\mathrm{gap}},\, 1,\, 2\Delta_\phi - 1)$ (\texttt{exists\_const\_abs\_pairSpin\_sub\_universal\_le}). Both theorems are proven with zero \texttt{sorry}'s and no assumptions beyond Assumption~\ref{assump:base}.

\section{Quantitative Primary Lightcone Bootstrap for All \texorpdfstring{$\Delta_\phi > 0$}{Delta\_phi > 0}}

In the monomial expansion of Section~2, each primary operator of spin $\ell$ and twist $h = \frac{\Delta - \ell}{2}$ is dressed by its tower of $SL(2, \mathbb{R})$ lightcone descendants, and the $\Delta_\phi > 1/2$ restriction arises because the $t$-channel identity $(1 - e^{w_r})^{-2\Delta_\phi}$ has spin-$2k$ Fourier projection of order $(2k)^{2\Delta_\phi - 1}$, which only dominates the $O(1)$ outer contour when $\Delta_\phi > 1/2$. To extract the $d$-dimensional primary operator measures $\mu^{\mathrm{prim}}_\ell$ directly and eliminate the $\Delta_\phi > 1/2$ barrier, we act on the right coordinate $w_r \in \mathbb{H}_-$ with the 1D fractional transmutation operator $\mathcal{L}_0 f(w_r) := \frac{1}{\sqrt{\pi}}(1 - e^{w_r})\partial_{w_r}\bigl[(1 - e^{w_r})\partial_{w_r} - \tfrac{1}{2}(1 + e^{w_r})\bigr] \int_0^1 x^{-1/2}(1-x)^{-1/2}(1 - x e^{2 w_r})^{-1/2} f(w_r + \log x)\, dx$ (\texttt{Transmute1d.lean}). Carrying the derivatives inside the integral and integrating by parts twice on $x \in (0, 1)$ gives the regularised inner-derivative representation (\texttt{transmuteOpInner} in \texttt{TransmutedEndpointBounds.lean}):
\begin{equation}
  \label{eq:L0-def}
  (\mathcal{L}_0 f)(w_r) = \frac{1}{\sqrt{\pi}} \int_0^1 x^{-1/2}(1 - x)^{-1/2}\bigl(1 - x e^{2 w_r}\bigr)^{-1/2} \, \mathcal{D}_{\mathrm{inner}}(w_r, x, f) \, dx,
\end{equation}
where $\mathcal{D}_{\mathrm{inner}}(w_r, x, f) := (1 - e^{w_r})^2 \bigl[\frac{1 + x e^{w_r}}{1 - x e^{w_r}} f''(w_r + \log x) + \frac{2 x e^{w_r}}{(1 - x e^{w_r})^2} f'(w_r + \log x)\bigr] - \frac{1}{2}(1 + e^{w_r})^2 f'(w_r + \log x)$.
We prove in \texttt{TransmutedEndpointBounds.lean} (zero \texttt{sorry}'s) that $\mathcal{L}_0$ annihilates the $s$-channel vacuum $k_0 = 1$ (\texttt{transmuteOpInner\_const}) and maps every 1D collinear block $k_{\bar{h}}(e^{w_r}) := e^{\bar{h} w_r} \, {}_2F_1\bigl(\bar{h}, \tfrac{1}{2}; \bar{h} + \tfrac{1}{2}; e^{2 w_r}\bigr)$ for all $\bar{h} > 0$ identically to a single exponential monomial (\texttt{transmuteOpInner\_block1d}):
\begin{equation}
  \label{eq:L0-block1d}
  \mathcal{L}_0\bigl[k_{\bar{h}}\bigr](w_r) = w_0(\bar{h}) \, e^{\bar{h} w_r}, \qquad
  w_0(\bar{h}) := \Bigl(\bar{h} - \frac{1}{2}\Bigr) \frac{\Gamma(\bar{h} + 1/2)}{\Gamma(\bar{h})} \quad (\bar{h} > 0).
\end{equation}
Expanding the $t$-channel identity factor into 1D collinear blocks via $\mathcal{C}_{\Delta_\phi}(w_r) = \sum_{m=0}^\infty c_m(\Delta_\phi) \, k_{\Delta_\phi + m}(e^{w_r})$ with $c_m(\Delta_\phi) := 4^{\Delta_\phi} \frac{(2\Delta_\phi)_m (\Delta_\phi)_m}{m! (2\Delta_\phi + m - 1)_m}$ and applying~\eqref{eq:L0-block1d} termwise (\texttt{TransmutedCrossing.lean}) yields the exact 3-term transformed reference $S_0(w_r) = \frac{4^{\Delta_\phi} \Gamma(2\Delta_\phi + 1)}{\Gamma(\Delta_\phi)^2} \sum_{m=0}^\infty \mathfrak{m}_m(\Delta_\phi) \, e^{(\Delta_\phi + m) w_r}$ (\texttt{crossFactorExactTransmuted} in \texttt{ExactTransmutedRef.lean}):
\begin{align}
  \label{eq:S0-def}
  S_0(w_r) &:= \frac{4^{\Delta_\phi} \Gamma(2\Delta_\phi + 1)}{\Gamma(\Delta_\phi)^2} e^{\Delta_\phi w_r} \bigl(1 - e^{w_r}\bigr)^{-(2\Delta_\phi + 1)} \left[1 - (1 - e^{w_r}) + \frac{\Delta_\phi - 1/2}{4\Delta_\phi}(1 - e^{w_r})^2\right], \\
  \mathfrak{m}_m(\Delta_\phi) &:= \frac{(2\Delta_\phi + 1)_m}{m!} - \frac{(2\Delta_\phi)_m}{m!} + \frac{\Delta_\phi - 1/2}{4\Delta_\phi} \frac{(2\Delta_\phi - 1)_m}{m!} = \frac{(\Delta_\phi + m - 1/2)^2}{2\Delta_\phi \, m} \frac{(2\Delta_\phi)_{m-1}}{(m-1)!} > 0 \quad (m \ge 1),
\end{align}
with $\mathfrak{m}_0(\Delta_\phi) = \frac{\Delta_\phi - 1/2}{4\Delta_\phi}$.

\begin{assumption}[Collinear Block Decomposition --- \texttt{Blocks\_pdisk.lean}, \texttt{LightconeBlockDecomp.lean}, \texttt{TransmutedCrossing.lean}]
\label{assump:lightcone-blocks}
Each $d$-dimensional conformal block $G_{\Delta, \ell}(w_l, w_r)$ decomposes at left-moving twist levels $h + m = \frac{\Delta - \ell}{2} + m$ ($m \in \mathbb{N}$) into finite non-negative linear combinations of $m + 1$ collinear blocks $e^{(h + m) w_l} k_{h + \ell + j}(e^{w_r})$ with $j = m - 2s$ ($0 \le s \le m$), where the leading twist level $m = 0$ is a single 1D block $C_{0, \ell} \, e^{h w_l} k_{h + \ell}(e^{w_r})$ with $C_{0, \ell} = \frac{(\nu)_\ell}{\ell!} > 0$ (\texttt{ConformalBlock.toLightcone} in \texttt{Blocks\_pdisk.lean}). Pushing $d\mu_{\mathrm{OPE}}(\Delta, \ell)$ forward through this expansion defines non-negative 1D lightcone-block spectral measures $\{\mu^{\mathrm{LC}}_J\}_{J \in 2\mathbb{N}}$ on $[h_{\mathrm{gap}}, \infty)$ (\texttt{FourPointData.lightconeMeasure}) that expand the correlator as
\begin{equation}
  \label{eq:collinear-decomp}
  \mathcal{G}(w_l, w_r) = 1 + \sum_{k=0}^\infty \int_{[h_{\mathrm{gap}}, \infty)} e^{h w_l} k_{h + 2k}\bigl(e^{w_r}\bigr) \, d\mu^{\mathrm{LC}}_{2k}(h).
\end{equation}
\end{assumption}
By Fubini and dominated convergence on $s \in (0, 1)$, we prove in \texttt{TransmutedCrossing.lean} (zero \texttt{sorry}'s, \texttt{transmuteOp\-Inner\_evalCollinear} and \texttt{transmuteOp\-Inner\_gffBlockSeries}) that $\mathcal{L}_0$ commutes with both the $s$-channel sum and integral in~\eqref{eq:collinear-decomp} and the 1D GFF block series of $\mathcal{C}_{\Delta_\phi}(w_r)$, giving $\mathcal{L}_0[\mathcal{C}_{\Delta_\phi}](w_r) = S_0(w_r)$ and $\mathcal{L}_0[\mathcal{G}(w_l, \cdot)](w_r) = \sum_{k=0}^\infty e^{2k w_r} \!\int_{[h_{\mathrm{gap}}, \infty)} w_0(h + 2k) e^{h(w_l + w_r)} d\mu^{\mathrm{LC}}_{2k}(h)$ (\texttt{toLightconeBlockFourPointData\_agreesWithCorrelator}).

\begin{theorem}[Quantitative 1D Collinear Lightcone Bootstrap for All \texorpdfstring{$\Delta_\phi > 0$}{Delta\_phi > 0} --- \texttt{TransmutedCrossing.lean}, \texttt{ExactTransmutedRef.lean}]
\label{thm:primary-bootstrap}
Suppose Assumptions~\ref{assump:base} and~\ref{assump:lightcone-blocks} hold for $\nu \ge 0$, external dimension $\Delta_\phi > 0$, and twist gap $h_{\mathrm{gap}} > 0$. Fix $N \in \mathbb{N}$ and any target exponent $0 < \gamma < \min\bigl(2 h_{\mathrm{gap}},\, 2\Delta_\phi\bigr)$. Then there exists $C_N > 0$ such that for all integers $k \ge 1$ and all $\varphi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{h \in \mathbb{R}} \max\{|\varphi(h)|, |\varphi'(h)|, |\varphi''(h)|\} \le M_\varphi$,
\begin{multline}
  \label{eq:primary-bootstrap}
  \Biggl|(2k)^{-2\Delta_\phi} \int_{\mathbb{R}} w_0(h + 2k) \, \varphi(h) \, d\mu^{\mathrm{LC}}_{2k}(h) \\
    - (2k)^{-2\Delta_\phi} \frac{2 \cdot 4^{2\Delta_\phi} \Gamma(2\Delta_\phi + 1)}{\Gamma(\Delta_\phi)^2} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \, \mathfrak{m}_{n + 2k}(\Delta_\phi) \, \varphi(\Delta_\phi + n)\Biggr|
  \le C_N \, (2k)^{-\gamma} \, M_\varphi.
\end{multline}
\end{theorem}

\paragraph{Proof sketch and large-$k$ expansion of the universal part.}
Along the folded contour $w_r(\theta) = -1/(2k) + i\theta$, $w_l(\theta) = \sigma + 1/(2k) - i\theta$ ($\theta \in [-\pi/2, \pi/2]$), two-variable crossing~\eqref{eq:two-var-crossing} expresses the untransmuted remainder as $R_\theta(z) := \mathcal{G}(w_l(\theta), z) - \mathcal{C}_{\Delta_\phi}(w_l(\theta))\mathcal{C}_{\Delta_\phi}(z) = \mathcal{C}_{\Delta_\phi}(w_l(\theta))\mathcal{C}_{\Delta_\phi}(z)\bigl(\mathcal{G}(w_l'(\theta), z') - 1\bigr)$ (\texttt{TransmutedCrossing.lean}). On the $s$-channel portion $x \in (0, 1/2]$, $\operatorname{Re}(w_r(\theta) + \log x) < -1/2$, so Cauchy's estimate gives uniform derivative bounds $\|R_\theta^{(j)}(w_r(\theta) + \log x)\| \le M_j$ ($j = 1, 2$). On the $t$-channel portion $x \in (1/2, 1)$, $\operatorname{Re} z' \le \sigma_{r, 0} < 0$ stays uniformly inside $\mathbb{H}_-$, so Cauchy's estimate on a fixed-radius disk around $z'$ combined with the explicit derivatives of $\mathcal{C}_{\Delta_\phi}(z)$ and $z'(z) = 2\log\frac{1 - e^{z/2}}{1 + e^{z/2}}$ gives $\|R_\theta^{(j)}(w_r(\theta) + \log x)\| \le C_j \bigl(\|1 - e^{w_r(\theta)}\| + 1 - x\bigr)^{2(h' - \Delta_\phi) - j}$ for any effective gap $0 < h' \le h_{\mathrm{gap}}$. The factor $(1 - e^{w_r})^2$ in $\mathcal{D}_{\mathrm{inner}}$ cancels two inverse powers of $\|1 - x e^{w_r}\| \ge \frac{1}{3}(\|1 - e^{w_r}\| + 1 - x)$, and integrating the resulting Beta tail $(1 - x)^{-1/2}(\|1 - e^{w_r}\| + 1 - x)^{2(h' - \Delta_\phi) - 3/2}$ over $x \in (1/2, 1)$ and $\theta \in [-\pi/2, \pi/2]$ bounds the folded integral by $O\bigl((2k)^{-2 h'} + (2k)^{-2\Delta_\phi}\bigr)$ (\texttt{Transmuted\-Endpoint\-Bounds.lean}). Choosing $h' = \frac{1}{2}(\gamma/2 + \min(h_{\mathrm{gap}}, \Delta_\phi)) < \Delta_\phi$ avoids logarithmic resonances at $2(h' - \Delta_\phi) \in \{0, 1\}$, while 2-step integration by parts on $[-\pi, -\pi/2] \cup [\pi/2, \pi]$ bounds the outer-contour discrepancy of $S_0(w_r)$ by $O\bigl((2k)^{-2\Delta_\phi}\bigr)$ (\texttt{ExactTransmutedRef.lean}). Expanding $\mathfrak{m}_{n + 2k}(\Delta_\phi) = \frac{(2\Delta_\phi + 1)_{n + 2k}}{(n + 2k)!} + O(k^{-1}) = \frac{(2k)^{2\Delta_\phi}}{\Gamma(2\Delta_\phi + 1)} + O(k^{-1})$ at large $k$ reduces the subtracted sum in~\eqref{eq:primary-bootstrap} to either the 1-term binomial sum $\frac{(2\Delta_\phi + 1)_{n + 2k}}{(n + 2k)!}$ or the leading Gamma limit $\frac{2 \cdot 4^{2\Delta_\phi}}{\Gamma(\Delta_\phi)^2} \sum_{n=0}^{\infty} \frac{(2\Delta_\phi)_n}{n!} \, \varphi(\Delta_\phi + n) + O(k^{-1})$.

\begin{assumption}[Normalized Subleading Conformal Block Coefficients --- \texttt{PrimaryAllTwists.lean}]
\label{assump:collinear-transfer}
At left-moving twist level $m \ge 1$ and collinear spin shift $j = m - 2s$ ($0 \le s \le m$), a $d$-dimensional primary of positive twist $h > 0$ and spin $\ell = 2(k + s)$ contributes $B_{m, m - 2s}(h, 2(k + s)) \, e^{(h + m)w_l} k_{h + m + 2k}(e^{w_r})$ to the 1D collinear spin-$2k$ channel, where the normalized conformal block ratios for $h > 0$
\begin{equation}
  \label{eq:normalized-block-coeffs}
  \mathcal{R}_{m, s}(\nu, k, h) := \frac{B_{m, m - 2s}(h, 2(k + s))}{C_{0, 2(k + s)}}
\end{equation}
(\texttt{normalizedCollinearBlockCoeff}) are $C^2$ in $h \in (0, \infty)$ with $\sup_{k \ge 1} \sup_{0 < h \le N} \max_{0 \le r \le 2} |\partial_h^r \mathcal{R}_{m, s}(\nu, k, h)| < \infty$ for each fixed $(m, s, N)$.\footnote{The $C^2$ regularity, uniform bounds, and finite triangular decomposition of $\mathcal{R}_{m, s}$ (\texttt{normLCPairing\_eq\_\allowbreak normPrimaryPairing\_add\_sum} in \texttt{PrimaryAllTwists.lean}), together with \texttt{exists\_conformalBlockLightcone} in \texttt{Blocks\_pdisk.lean} and the collinear sum-integral convergence hypotheses in \texttt{toLightconeBlockFourPointData} and \texttt{toLightconeBlockFourPointData\_agreesWithCorrelator}, are properties of the collinear conformal block expansion stated with \texttt{sorry}; all analytical and Tauberian deductions from them are proven with zero \texttt{sorry}'s.}
\end{assumption}
Multiplying $\mathcal{R}_{m, s}(\nu, k, h)$ by the explicit rational ratio of transmutation weights yields the transmuted transfer ratio for $h > 0$
\begin{equation}
  \label{eq:transfer-kernel}
  \mathcal{K}_{m, s}(\nu, \Delta_\phi, k, h) := \left(\frac{k + s}{k}\right)^{2\Delta_\phi} \frac{w_0(h + m + 2k)}{w_0(h + 2(k + s))} \, \mathcal{R}_{m, s}(\nu, k, h),
\end{equation}
which we prove in \texttt{PrimaryAllTwists.lean} (zero \texttt{sorry}'s, \texttt{collinearTransferFn}) has $\sup_{k \ge 1} \sup_{0 < h \le N} \max_{0 \le r \le 2} |\partial_h^r \mathcal{K}_{m, s}(\nu, \Delta_\phi, k, h)| < \infty$ for each $(m, s, N)$ and extends smoothly by zero below $\frac{1}{2}\min(h_{\mathrm{gap}}, \Delta_\phi) > 0$ (where neither the primary nor the GFF measure has support).

\begin{theorem}[All-Twist $d$-Dimensional Primary Lightcone Bootstrap --- \texttt{PrimaryAllTwists.lean}]
\label{thm:all-twist-primary}
Suppose Assumptions~\ref{assump:base}, \ref{assump:lightcone-blocks}, and~\ref{assump:collinear-transfer} hold for $\nu \ge 0$, $\Delta_\phi > 0$, and $h_{\mathrm{gap}} > 0$, and define the universal finite-spin Generalized Free Field (GFF) $d$-dimensional primary coefficients $c^{\mathrm{GFF, prim}}_n(\nu, \Delta_\phi, k)$ recursively over $n \in \mathbb{N}$ by
\begin{multline}
  \label{eq:gff-prim-rec}
  c^{\mathrm{GFF, prim}}_n(\nu, \Delta_\phi, k) := (2k)^{-2\Delta_\phi} \frac{2 \cdot 4^{2\Delta_\phi} \Gamma(2\Delta_\phi + 1)}{\Gamma(\Delta_\phi)^2} \frac{(2\Delta_\phi)_n}{n!} \, \mathfrak{m}_{n + 2k}(\Delta_\phi) \\
  - \sum_{m=1}^{n} \sum_{s=0}^{m} \mathcal{K}_{m, s}(\nu, \Delta_\phi, k, \Delta_\phi + n - m) \, c^{\mathrm{GFF, prim}}_{n - m}(\nu, \Delta_\phi, k + s).
\end{multline}
Fix $N \in \mathbb{N}$ and any $0 < \gamma < \min\bigl(2 h_{\mathrm{gap}},\, 2\Delta_\phi\bigr)$. Then there exists $C_N > 0$ such that for all integers $k \ge 1$ and all $\varphi \in C^2(\mathbb{R})$ supported in $[-N, N]$ with $\sup_{h \in \mathbb{R}} \max\{|\varphi(h)|, |\varphi'(h)|, |\varphi''(h)|\} \le M_\varphi$,
\begin{equation}
  \label{eq:all-twist-primary}
  \Biggl|(2k)^{-2\Delta_\phi} C_{0, 2k} \int_{\mathbb{R}} w_0(h + 2k) \, \varphi(h) \, d\mu^{\mathrm{prim}}_{2k}(h)
    - \sum_{n=0}^{\infty} c^{\mathrm{GFF, prim}}_n(\nu, \Delta_\phi, k) \, \varphi(\Delta_\phi + n)\Biggr|
  \le C_N \, (2k)^{-\gamma} \, M_\varphi.
\end{equation}
\end{theorem}

\paragraph{Proof sketch.}
For any test function $\varphi$ vanishing on $[B_0 + M + 1, \infty)$ ($M \in \mathbb{N}$), decomposing $\mu^{\mathrm{LC}}_{2k}$ by left-moving twist level $m \ge 0$ expresses the normalized primary pairing $\mathcal{P}_k(\varphi) := (2k)^{-2\Delta_\phi} C_{0, 2k} \int w_0(h + 2k) \varphi(h) \, d\mu^{\mathrm{prim}}_{2k}(h)$ as the finite triangular system $\mathcal{P}_k(\varphi) = \mathcal{L}_k(\varphi) - \sum_{m=1}^{M} \sum_{s=0}^{m} \mathcal{P}_{k + s}(\psi_{m, s, k})$, where $\psi_{m, s, k}(h) := \mathcal{K}_{m, s}(k, h)\varphi(h + m)$ vanishes outside $[-N, N]$ and on $[B_0 + M, \infty)$, with $C^2$ norm bounded uniformly in $k$ by $4 B_{m, s, N} M_\varphi$. By~\eqref{eq:gff-prim-rec}, the GFF sum $\sum_{n=0}^\infty c^{\mathrm{GFF, prim}}_n(\Delta_\phi, k)\varphi(\Delta_\phi + n)$ satisfies the identical triangular relation with $\mathcal{L}_k(\varphi)$ replaced by the subtracted sum of Theorem~\ref{thm:primary-bootstrap} (\texttt{gffPrimaryPairing\_eq\_exactIdentity\_sub\_sum}). Induction on $M \in \mathbb{N}$ using Theorem~\ref{thm:primary-bootstrap} and $(2(k + s))^{-\gamma} \le (2k)^{-\gamma}$ ($s \ge 0$) establishes~\eqref{eq:all-twist-primary} on $[-N, N] \cap (-\infty, B_0 + M + 1)$ for every $M$, and setting $M = N$ covers all of $[-N, N]$; dividing $\varphi(h)$ by $w_0(h + 2k)/(2k)^{1/2}$ similarly yields the unweighted primary pairing bound (\texttt{PrimaryAllTwists.lean}).

\section{Quantitative High-Energy Primary Asymptotics (\texorpdfstring{$\Delta \to \infty$}{Delta -> infty})}

Whereas Sections~2 and~3 analyze the large-spin lightcone limit ($\ell \to \infty$ at fixed twist $h$, governed by $w_l \to -\infty, w_r \to 0^-$), the module \texttt{ConeTauberian/} analyzes the bulk high-energy limit $\Delta \to \infty$ as a joint distribution in the rescaled scaling dimension $u = \Delta'/\Delta \in (0, \infty)$ and spin-to-dimension ratio $\xi = \ell/\Delta' \in (0, 1)$, governed by $(w_l, w_r) \to (0, 0)$. Applying $\mathcal{L}_0$ in both coordinates $(w_l, w_r)$ (\texttt{Branching2d.lean}, zero \texttt{sorry}'s) maps every 2D conformal block $G^{(2\mathrm{d})}_{\Delta', j}(w_l, w_r) := \frac{1}{1 + \delta_{j, 0}}\bigl[k_{(\Delta'-j)/2}(e^{w_l}) k_{(\Delta'+j)/2}(e^{w_r}) + (w_l \leftrightarrow w_r)\bigr]$ to a pure symmetrized exponential weighted by $w_0(\frac{\Delta'-j}{2}) w_0(\frac{\Delta'+j}{2})$.

\begin{assumption}[Bilateral Transmuted Crossing and $d \to 2$ Branching Reduction --- \texttt{ConeAsymptotics.lean}]
\label{assump:cone}
Let $d = 2\nu + 2 \ge 3$ ($\nu > 0$) and let $\mu^{\mathrm{trans}, 2\mathrm{d}}_\Delta$ (\texttt{FourPointData.transmutedMeasure2d}) and $\mu^{\mathrm{id, trans}, 2\mathrm{d}}_\Delta$ (\texttt{transmutedIdentityMeasure2d}) denote the rescaled 2D bilaterally transmuted OPE and $t$-channel identity measures on $(u_l, u_r) = (h'/\Delta, \bar{h}'/\Delta) \in (0, \infty)^2$, normalized by $\Delta^{-2 - 4\Delta_\phi}$.
\begin{enumerate}
  \item \emph{Bilateral transmuted crossing bound}: At complex rescaled points $w_{l,r} = (-1 + i t_{l,r})/\Delta$ with $(t_l, t_r) \in [-R, R]^2$, the 2D Fourier--Laplace transforms of $\mu^{\mathrm{trans}, 2\mathrm{d}}_\Delta$ and $\mu^{\mathrm{id, trans}, 2\mathrm{d}}_\Delta$ are uniformly bounded at $t_l = t_r = 0$ and agree on $[-R, R]^2$ up to the non-identity $t$-channel remainder $O\bigl(R^{4\Delta_\phi} \Delta^{-2 h_{\mathrm{gap}}}\bigr)$ for all $\Delta \ge 1$ (\texttt{exists\_laplace2d\_sub\_identity\_bound}).
  \item \emph{Universal $O(\Delta^{-1})$ $d \to 2$ branching and Stirling reduction}: For any smooth test function $\chi(u, \xi)$ compactly supported in $(0, \infty) \times (0, 1)$, applying the inverse Erd\'elyi--Kober fractional integral in $\xi$ and dividing by $w_0(h') w_0(\bar{h}')$ produces a 2D test function $\psi(u_l, u_r)$ with integrable Fourier decay such that both the discrete $d \to 2$ conformal-block branching error between $\Delta^{2\nu - 4\Delta_\phi} \int \chi(\Delta'/\Delta, \ell/\Delta') \, d\mu_{\mathrm{OPE}}(\Delta', \ell)$ and $\int \psi \, d\mu^{\mathrm{trans}, 2\mathrm{d}}_\Delta$, and the universal Stirling error between $\int \psi \, d\mu^{\mathrm{id, trans}, 2\mathrm{d}}_\Delta$ and the continuum integral~\eqref{eq:cone-bootstrap}, are $O(\Delta^{-1})$ (\texttt{exists\_test2d\_of\_smooth\_compactSupport}).\footnote{The underlying $O(\Delta^{-1})$ Stirling bounds for the Gegenbauer branching coefficients $b_m^{(\nu)}(\ell)$ and transmutation weights $w_0$, as well as the exact Erd\'elyi--Kober Beta integral identity mapping $\rho^{\mathrm{prim}}_{\nu, \Delta_\phi}$ to the 2D effective density, are proven with zero \texttt{sorry}'s in \texttt{Branching2d.lean} and \texttt{ErdelyiKoberInversion.lean}.}
\end{enumerate}
\end{assumption}

\begin{theorem}[Quantitative High-Energy Primary Cone Bootstrap --- \texttt{ConeAsymptotics.lean}]
\label{thm:cone-bootstrap}
Let $\nu > 0$, $\Delta_\phi > \frac{\nu}{2} + \frac{1}{4}$, and $h_{\mathrm{gap}} > 0$, and define the continuum primary spectral density on $(u, \xi) \in (0, \infty) \times (0, 1)$ by
\begin{equation}
  \rho^{\mathrm{prim}}_{\nu, \Delta_\phi}(u, \xi) := c_{\nu, \Delta_\phi} \, u^{4\Delta_\phi - 2\nu - 1} \, \xi^{2\nu} \bigl(1 - \xi^2\bigr)^{2\Delta_\phi - \nu - 3/2}, \qquad
  c_{\nu, \Delta_\phi} := \frac{2^{2\nu + 1} \, \Gamma(\nu) \, \Gamma(2\Delta_\phi - 1/2)}{\Gamma(\Delta_\phi)^4 \, \Gamma(2\Delta_\phi - \nu - 1/2)}.
\end{equation}
Under Assumption~\ref{assump:cone}, for any smooth test function $\chi$ compactly supported in $(0, \infty) \times (0, 1)$, set $\gamma := \min\bigl(\frac{2 h_{\mathrm{gap}}}{4\Delta_\phi + 1},\, 1\bigr) > 0$. Then there exists $C_\chi > 0$ such that for all $\Delta \ge 1$,
\begin{equation}
  \label{eq:cone-bootstrap}
  \left|\Delta^{2\nu - 4\Delta_\phi} \int \chi\!\left(\frac{\Delta'}{\Delta},\, \frac{\ell}{\Delta'}\right) d\mu_{\mathrm{OPE}}(\Delta', \ell)
    - \int_0^\infty \int_0^1 \chi(u, \xi) \, \rho^{\mathrm{prim}}_{\nu, \Delta_\phi}(u, \xi) \, d\xi \, du\right|
  \le C_\chi \, \Delta^{-\gamma}.
\end{equation}
\end{theorem}

\paragraph{Proof sketch and related results.}
Because $w_{l,r} = (-1 + i t_{l,r})/\Delta \to 0$ as $\Delta \to \infty$ for all $|t_{l,r}| \le R = \Delta^\gamma$, the evaluation stays inside the $t$-channel OPE convergence neighborhood without any outer contour. A bandlimited 2D Fourier--Laplace Tauberian theorem (\texttt{Karamata2d.lean}, zero \texttt{sorry}'s) with frequency cutoff $R = \Delta^{2 h_{\mathrm{gap}} / (4\Delta_\phi + 1)}$ first establishes a pure $O\bigl(\Delta^{-2 h_{\mathrm{gap}}/(4\Delta_\phi + 1)}\bigr)$ bound between $\int \psi \, d\mu^{\mathrm{trans}, 2\mathrm{d}}_\Delta$ and $\int \psi \, d\mu^{\mathrm{id, trans}, 2\mathrm{d}}_\Delta$ valid for all $\Delta_\phi > 0$. In the continuum limit, the $d \to 2$ conformal-block branching coefficients $b_m^{(\nu)}(\ell)$ converge to the Erd\'elyi--Kober fractional integral kernel $\mathcal{K}_\nu(\xi, \xi') = \frac{2^{2-2\nu}}{\Gamma(\nu)^2}\xi^{1-2\nu}(\xi^2 - (\xi')^2)^{\nu-1}\mathbf{1}_{\xi' < \xi}$, which is inverted via Euler's Beta integral (\texttt{ErdelyiKoberInversion.lean}) to yield $\rho^{\mathrm{prim}}_{\nu, \Delta_\phi}(u, \xi)$ up to the universal $O(\Delta^{-1})$ Stirling error. Finally, sandwiching the indicator of the unit cylinder $(u, \xi) \in (0, 1] \times (0, 1)$ by smooth test functions (\texttt{exists\_smooth\_sandwich\_unitCylinder}) yields the cumulative spin-summed primary asymptotic (\texttt{FourPointData.tendsto\_cumulativePrimaryWeight}) as $\Delta \to \infty$:
\begin{equation}
  \int_{0 < \Delta' \le \Delta} d\mu_{\mathrm{OPE}}(\Delta', \ell) \;\underset{\Delta \to \infty}{\sim}\;
  \frac{2^{2\nu - 1} \, \Gamma(\nu) \, \Gamma(\nu + 1/2) \, \Gamma(2\Delta_\phi - 1/2)}{(2\Delta_\phi - \nu) \, \Gamma(\Delta_\phi)^4 \, \Gamma(2\Delta_\phi)} \, \Delta^{4\Delta_\phi - 2\nu}.
\end{equation}
